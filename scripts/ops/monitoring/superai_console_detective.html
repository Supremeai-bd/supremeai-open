📝 The Capture Script (Copy This)

Save this as superai_console_capture.js or paste directly: 

// SuperAI Console Capture v1.0
// Paste this ENTIRE script into any website's F12 console

(function() {
    window.__superai_logs = [];
    
    const _orig = { 
        log: console.log.bind(console), 
        error: console.error.bind(console),
        warn: console.warn.bind(console),
        info: console.info.bind(console)
    };
    
    function capture(level, args) {
        window.__superai_logs.push({
            time: new Date().toISOString(),
            level: level,
            args: Array.from(args).map(String)
        });
    }
    
    console.log = function(...a) { capture('log', a); _orig.log(...a); };
    console.error = function(...a) { capture('error', a); _orig.error(...a); };
    console.warn = function(...a) { capture('warn', a); _orig.warn(...a); };
    console.info = function(...a) { capture('info', a); _orig.info(...a); };
    
    window.addEventListener('error', e => {
        capture('uncaught', [e.message, e.filename + ':' + e.lineno]);
    });
    
    window.downloadSuperAILogs = function() {
        const data = {
            url: location.href,
            exported: new Date().toISOString(),
            count: window.__superai_logs.length,
            logs: window.__superai_logs
        };
        
        const blob = new Blob([JSON.stringify(data, null, 2)], {type:'application/json'});
        const a = document.createElement('a');
        a.href = URL.createObjectURL(blob);
        a.download = 'console_logs_' + Date.now() + '.json';
        a.click();
        
        console.log('%c✅ Downloaded ' + data.count + ' log entries!', 'color:green;font-size:14px');
    };
    
    console.log('%c🔍 SuperAI Console Capture Active!', 'color:#00d2ff;font-size:16px;font-weight:bold');
    console.log('Type downloadSuperAILogs() to export');
})();
